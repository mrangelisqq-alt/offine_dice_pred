<script>
const COLORS = ['Red', 'Orange', 'Yellow', 'Green', 'Blue', 'Purple'];
const COLOR_HEX = { Red: '#ff0000', Orange: '#ffa500', Yellow: '#ffff00', Green: '#00ff00', Blue: '#0000ff', Purple: '#800080' };

const WSS_URL = "wss://s-gke-euw1-nssi3-4.europe-west1.firebasedatabase.app/.ws?v=5&p=1:593264096902:web:ade81a908d70bdf7d90307&ns=offline-dice-verify-default-rtdb";

let currentSession = null;
let ws = null;
// Firma dell'ultimo roll processato: usiamo ID + colori per evitare
// di perdere roll nuovi quando Firebase riusa gli stessi path/ID.
let lastProcessedSignature = null;

function log(msg) {
  const el = document.getElementById('log');
  const d = document.createElement('div');
  d.textContent = `[${new Date().toLocaleTimeString()}] ${msg}`;
  el.prepend(d);
}

function setStatus(msg) {
  document.getElementById('status').textContent = 'Status: ' + msg;
}

function connect() {
  const sid = document.getElementById('sessionId').value.trim();
  if (!sid) { setStatus('ERROR: No session ID'); return; }

  if (ws) { ws.close(); ws = null; }

  currentSession = sid;
  lastProcessedSignature = null;
  document.querySelectorAll('.die').forEach(d => {
    d.classList.remove('active', 'predicted');
    const c = d.querySelector('.color');
    c.textContent = '—';
    c.style.color = '';
  });

  setStatus('Connecting to WebSocket...');
  log(`Connecting to Firebase WSS for session ${sid}`);

  ws = new WebSocket(WSS_URL);

  ws.onopen = () => {
    setStatus('Connected - Authenticating');
    log('WSS stream OPEN. Waiting for handshake.');
  };

  ws.onmessage = (event) => {
    // Debug: decommenta la riga sotto per vedere TUTTO quello che manda Firebase
    // console.log('RAW WSS:', event.data);

    try {
      const data = JSON.parse(event.data);

      // --- Handshake Firebase ---
      if (data.t === 'c' && data.d && data.d.t === 'h') {
        log('Handshake received. Sending LISTEN command.');
        setStatus('Listening (real-time)');

        const listenCmd = {
          t: "d",
          d: {
            r: 1,
            a: "q",
            b: {
              p: `/sessions/${currentSession}/rolls`,
              h: ""
            }
          }
        };
        ws.send(JSON.stringify(listenCmd));
        log(`Sent Firebase LISTEN for /sessions/${currentSession}/rolls`);
        return;
      }

      // --- Dati ---
      if (data.t === 'd' && data.d) {
        const b = data.d.b;
        const p = data.d.p || '';

        if (!b) return;

        // Gestisce un singolo oggetto roll, con ID esplicito
        const tryProcess = (id, obj) => {
          if (!obj || !obj.names) return;

          const keys = Object.keys(obj.names).sort((a, b) => parseInt(a) - parseInt(b));
          const colors = keys.map(k => {
            const val = String(obj.names[k]).trim();
            const cap = val.charAt(0).toUpperCase() + val.slice(1).toLowerCase();
            return COLORS.includes(cap) ? cap : 'Unknown';
          });

          if (colors.length !== 4 || colors.includes('Unknown')) {
            return;
          }

          // Firma basata su ID + colori: così se il backend riusa lo stesso
          // ID ma cambiano i colori, lo processiamo comunque come nuovo roll.
          const signature = id + '|' + colors.join(',');
          if (signature === lastProcessedSignature) return;

          lastProcessedSignature = signature;
          displayRoll(colors);
        };

        // Caso A: riferimento diretto a un singolo roll
        // es. p = "/sessions/XXX/rolls/-Nxxx", b = { names: {...} }
        if (b.names && p.includes('/rolls/')) {
          const id = p.split('/rolls/')[1].split('/')[0];
          tryProcess(id, b);
          return;
        }

        // Caso B: mappa di roll (snapshot completo sotto /rolls)
        // es. b = { "-Nxxx": { names: {...} }, "-Nyyy": { names: {...} } }
        if (typeof b === 'object' && !b.names) {
          const ids = Object.keys(b).sort(); // gli ID push di Firebase sono ordinabili cronologicamente
          ids.forEach(id => tryProcess(id, b[id]));
          return;
        }

        // Caso C: oggetto singolo senza path esplicito
        if (b.names) {
          tryProcess('single', b);
          return;
        }
      }
    } catch (e) {
      log('Parse error: ' + e.message);
    }
  };

  ws.onerror = () => {
    log('WSS error occurred');
    setStatus('Error');
  };

  ws.onclose = () => {
    log('WSS stream CLOSED');
    if (document.getElementById('status').textContent.includes('Listening')) {
      setStatus('Disconnected');
    }
  };
}

function displayRoll(colors) {
  log('INTERCEPTED WINNING COLORS: [' + colors.join(', ') + ']');

  for (let i = 0; i < 4; i++) {
    const die = document.getElementById('die' + (i + 1));
    const c = die.querySelector('.color');
    c.textContent = colors[i];
    c.style.color = COLOR_HEX[colors[i]] || '#fff';
    die.classList.add('active', 'predicted');
  }

  setStatus('Current Roll: [' + colors.join(', ') + ']');
}
